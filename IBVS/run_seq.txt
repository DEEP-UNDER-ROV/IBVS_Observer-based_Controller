2. Cmd MAVROS
ros2 launch mavros apm.launch fcu_url:="serial:///dev/ttyACM0:921600" gcs_url:="udp://@192.168.4.1:14550"
ros2 launch mavros apm.launch fcu_url:="serial:///dev/ttyACM0:921600" gcs_url:="udp://@192.168.2.1:14550"


3. Cmd IBVS Parameter
ros2 run ibvs heave_trim
ros2 param set /heave_trim_node bias -133

ros2 param set /ibvs_rc_controller Kp [0.9,0.9,0.7,0.0,0.3,0.5]
ros2 param get /ibvs_rc_controller Kp [/Ki /Kd]


4. Cmd IBVS Controller
ros2 run ibvs visualize
ros2 run ibvs ibvs


5. Cmd Stream GStreamer
ros2 run ibvs stream
ros2 run ibvs stream --ros-args -p image_topic:="/camera/camera/color/image_raw"
ros2 run ibvs stream --ros-args -p image_topic:="/camera/infra1/image_rect_raw"
ros2 run ibvs stream --ros-args -p image_topic:="/corrected/left/image_raw"


6. Cmd Reset FCU
ros2 service call /mavros/cmd/command mavros_msgs/srv/CommandLong "{command: 246, param1: 1.0, param2: 0.0, param3: 0.0, param4: 0.0, param5: 0.0, param6: 0.0, param7: 0.0}"

ros2 service call /mavros/param/set mavros_msgs/srv/ParamSetV2 "{param_id: 'SYS_GCSID',value: {type: 2,integer_value: 255}}"
ros2 run mavros mavparam set SYS_GCSID 255
ros2 run mavros mavros_node --ros-args -p system_id:=255 -p target_system_id:=1
